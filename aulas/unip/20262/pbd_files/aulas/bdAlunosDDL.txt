USE Faculdade2026;

CREATE TABLE Alunos(
	ra VARCHAR(3) PRIMARY KEY,
    nome VARCHAR(10) NOT NULL
);

CREATE TABLE Cursos(
	id VARCHAR(3) PRIMARY KEY,
    nome VARCHAR(10) NOT NULL
);

CREATE TABLE Notas(
	id_nota INT AUTO_INCREMENT PRIMARY KEY,
    ra VARCHAR(3) NOT NULL,
    curso VARCHAR(3) NOT NULL,
    np1 DECIMAL(3, 1) NOT NULL,
    np2 DECIMAL(3, 1) NOT NULL,
    CONSTRAINT FK_Alunos FOREIGN KEY (ra) REFERENCES Alunos(ra),
    CONSTRAINT FK_Cursos FOREIGN KEY (curso) REFERENCES Cursos(id)
);
